#include <stdio.h>



int main() {

    int choice, quantity;

    int tshirtQty = 0, jeansQty = 0, shoesQty = 0, watchQty = 0;

    float total = 0;



    do {

        printf("\n===== ONLINE SHOPPING SYSTEM =====\n");

        printf("1. T-Shirt      - Rs 500\n");

        printf("2. Jeans        - Rs 1200\n");

        printf("3. Shoes        - Rs 2000\n");

        printf("4. Watch        - Rs 1500\n");

        printf("5. Exit\n");

        printf("Enter your choice: ");

        scanf("%d", &choice);



        switch(choice) {



            case 1:

                printf("Enter quantity of T-Shirts: ");

                scanf("%d", &quantity);

                tshirtQty += quantity;

                total += 500 * quantity;

                break;



            case 2:

                printf("Enter quantity of Jeans: ");

                scanf("%d", &quantity);

                jeansQty += quantity;

                total += 1200 * quantity;

                break;



            case 3:

                printf("Enter quantity of Shoes: ");

                scanf("%d", &quantity);

                shoesQty += quantity;

                total += 2000 * quantity;

                break;



            case 4:

                printf("Enter quantity of Watches: ");

                scanf("%d", &quantity);

                watchQty += quantity;

                total += 1500 * quantity;

                break;



            case 5:

                printf("\nThank you for shopping!\n");

                break;



            default:

                printf("Invalid choice! Please try again.\n");

        }



    } while(choice != 5);



    // Separate Bill

    printf("\n===== ITEM-WISE BILL =====\n");



    if(tshirtQty > 0)

        printf("T-Shirts  (%d x 500)  = Rs %d\n", tshirtQty, tshirtQty * 500);



    if(jeansQty > 0)

        printf("Jeans     (%d x 1200) = Rs %d\n", jeansQty, jeansQty * 1200);



    if(shoesQty > 0)

        printf("Shoes     (%d x 2000) = Rs %d\n", shoesQty, shoesQty * 2000);



    if(watchQty > 0)

        printf("Watches   (%d x 1500) = Rs %d\n", watchQty, watchQty * 1500);



    // Final Total

    printf("\n===== FINAL BILL =====\n");

    printf("Total Amount = Rs %.2f\n", total);



    return 0;

}
